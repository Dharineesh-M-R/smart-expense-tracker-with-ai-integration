// Displays the current date and time in the top right corner
void displayDateTime() {
    // Get local time
    time_t now = time(nullptr);
    struct tm* timeinfo = localtime(&now);

    // Format time string
    char timeStr[10];
    strftime(timeStr, sizeof(timeStr), "%H:%M:%S", timeinfo);
    
    // Format date string
    char dateStr[11];
    strftime(dateStr, sizeof(dateStr), "%d/%m/%Y", timeinfo);

    // Set text properties and calculate position
    tft.setTextSize(1);
    tft.setTextColor(TFT_CYAN, TFT_BLACK); // Set background to black to clear old text

    // Get the width of the text to align it to the right
    int timeWidth = tft.textWidth(timeStr);
    int dateWidth = tft.textWidth(dateStr);

    tft.setCursor(tft.width() - timeWidth - 5, 5);
    tft.print(timeStr);

    tft.setCursor(tft.width() - dateWidth - 5, 15);
    tft.print(dateStr);
}

void checkNumberPad(bool isPinPad) {
    static unsigned long lastTouch = 0;
    const int debounceDelay = 300;
    String* currentInput = isPinPad ? &currentPin : &amount;

    if (ts.touched()) {
        unsigned long currentTime = millis();
        if (currentTime - lastTouch < debounceDelay) return;
        lastTouch = currentTime;

        // Get and map touch point
        TS_Point p = ts.getPoint();
        int x = map(p.x, TS_MINX, TS_MAXX, 0, tft.width() - 1);
        int y = map(p.y, TS_MINY, TS_MAXY, 0, tft.height() - 1);
        x = constrain(x, 0, tft.width() - 1);
        y = constrain(y, 0, tft.height() - 1);

        int btnW = 60, btnH = 35;
        int startX = 30, startY = 80;
        int padding = 5;
        int okY = startY + 3 * (btnH + padding);
        String pressedNumber = "";

        // Check 1-9 buttons
        for (int row = 0; row < 3; row++) {
            for (int col = 0; col < 3; col++) {
                int btnX = startX + col * (btnW + padding);
                int btnY = startY + row * (btnH + padding);
                if (x > btnX && x < btnX + btnW && y > btnY && y < btnY + btnH) {
                    pressedNumber = String(row * 3 + col + 1);
                }
            }
        }

        // Check 0 button
        if (x > startX + (btnW + padding) && x < startX + (btnW + padding) + btnW &&
            y > okY && y < okY + btnH) {
            pressedNumber = "0";
        }
        
        // Check OK button
        if (x > startX && x < startX + btnW && y > okY && y < okY + btnH) {
            if (isPinPad) {
                // *** PIN CHECK AGAINST DB-FETCHED PIN ***
                if (*currentInput == cardPin) { 
                    pinEntered = true;
                    showCategories();
                    delay(300);
                } else {
                    // Display "Incorrect PIN!" over the number pad area
                    tft.fillRect(10, tft.height() - 30, tft.width() - 20, 20, TFT_BLACK);
                    tft.setCursor(10, tft.height() - 30);
                    tft.setTextColor(TFT_RED);
                    tft.setTextSize(2);
                    tft.print("Incorrect PIN!");
                    
                    // Add the desired 3-second (3000ms) delay
                    delay(3000); 

                    // Clear input and redraw the PIN pad for a new attempt
                    currentPin = "";
                    showNumberPad(true);
                }
                return;
            } else {
                if (amount.length() == 0) return;
                amountEntered = true;
                tft.fillScreen(TFT_BLACK);
                tft.setCursor(20, 20);
                tft.setTextColor(TFT_WHITE);
                tft.setTextSize(2);
                tft.print("Amount: $" + *currentInput);
                tft.println("\nTap NFC card...");
                return;
            }
        }

        // Check DEL button
        if (x > startX + 2 * (btnW + padding) && x < startX + 2 * (btnW + padding) + btnW &&
            y > okY && y < okY + btnH) {
            if (currentInput->length() > 0) currentInput->remove(currentInput->length() - 1);
            
            // Re-draw input display
            int inputLineY = isPinPad ? 35 : 20; // Adjusted Y for the amount display area
            tft.fillRect(30, inputLineY, 200, 30, TFT_BLACK);
            tft.setCursor(30, inputLineY + 5);
            tft.setTextColor(TFT_YELLOW);
            tft.setTextSize(3);

            String displayStr = *currentInput;
            if (isPinPad) {
                displayStr = "";
                for (size_t i = 0; i < currentInput->length(); i++) {
                    displayStr += "*";
                }
            }
            tft.print(displayStr);
            return;
        }

        // Handle number input
        if (pressedNumber != "") {
            if (isPinPad && currentInput->length() >= 4) return;
            if (!isPinPad && currentInput->length() >= 10) return;
            
            *currentInput += pressedNumber;
            
            // Re-draw input display
            int inputLineY = isPinPad ? 35 : 20; // Adjusted Y for the amount display area
            tft.fillRect(30, inputLineY, 200, 30, TFT_BLACK);
            tft.setCursor(30, inputLineY + 5);
            tft.setTextColor(TFT_YELLOW);
            tft.setTextSize(3);
            
            String displayStr = *currentInput;
            if (isPinPad) {
                displayStr = "";
                for (size_t i = 0; i < currentInput->length(); i++) {
                    displayStr += "*";
                }
            }
            tft.print(displayStr);
        }
    }
}

/**
 * @brief Fetches card, user name, and balance from Supabase.
 * Also performs the initial check if the entered amount is less than the balance.
 * * @return true if card/user found and balance is sufficient, false otherwise.
 */
bool getCardAndUserData() {
    if (WiFi.status() != WL_CONNECTED) {
        displayError("Wi-Fi Lost!");
        return false;
    }

    // 1. Fetch Card Details (user_id and pin)
    HTTPClient http;
    WiFiClientSecure client;
    client.setInsecure(); // Skip SSL cert verification

    // Filter by nfc_uid and select card_id, user_id, pin
    String cardUrl = "https://" + String(SUPABASE_HOST) + String(SUPABASE_URL_PREFIX) + "card?nfc_uid=eq." + cardUID + "&select=card_id,user_id,pin&limit=1"; // <<< MODIFIED
    http.begin(client, cardUrl);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("apikey", SUPABASE_API_KEY);
    http.addHeader("Authorization", "Bearer " + String(SUPABASE_API_KEY));

    int httpResponseCode = http.GET();
    String payload = "";

    if (httpResponseCode > 0) {
        payload = http.getString();
        Serial.println("Card Response: " + payload);
        
        // Simple JSON parsing to get user_id and pin
        if (payload.length() > 5 && payload != "[]") {
            // Find "card_id": "..." <<< NEW
            int cardIdStart = payload.indexOf("\"card_id\":\"") + 11;
            int cardIdEnd = payload.indexOf("\"", cardIdStart);
            currentCardID = payload.substring(cardIdStart, cardIdEnd);

            // Find "user_id": "..."
            int idStart = payload.indexOf("\"user_id\":\"") + 11;
            int idEnd = payload.indexOf("\"", idStart);
            currentUserID = payload.substring(idStart, idEnd);

            // Find "pin": "..."
            int pinStart = payload.indexOf("\"pin\":\"") + 7;
            int pinEnd = payload.indexOf("\"", pinStart);
            cardPin = payload.substring(pinStart, pinEnd);
            
            http.end();
        } else {
            http.end();
            displayError("Card Not Found!");
            return false;
        }
    } else {
        http.end();
        displayError("DB Error: " + String(httpResponseCode));
        return false;
    }
    
    // 2. Fetch User Name AND BALANCE
    String userUrl = "https://" + String(SUPABASE_HOST) + String(SUPABASE_URL_PREFIX) + "users?id=eq." + currentUserID + "&select=name,balance&limit=1"; // <<< MODIFIED: Added 'balance'
    http.begin(client, userUrl);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("apikey", SUPABASE_API_KEY);
    http.addHeader("Authorization", "Bearer " + String(SUPABASE_API_KEY));

    httpResponseCode = http.GET();
    
    if (httpResponseCode > 0) {
        payload = http.getString();
        Serial.println("User Response: " + payload);
        
        // Simple JSON parsing to get name and balance
        if (payload.length() > 5 && payload != "[]") {
            // Find "name": "..."
            int nameStart = payload.indexOf("\"name\":\"") + 8;
            int nameEnd = payload.indexOf("\"", nameStart);
            currentUserName = payload.substring(nameStart, nameEnd);
            
            // Find "balance": ...
            int balanceStart = payload.indexOf("\"balance\":") + 10;
            int balanceEnd = payload.indexOf(",", balanceStart);
            if (balanceEnd == -1) balanceEnd = payload.indexOf("]", balanceStart); // Handle case where balance is the last field in the array
            String balanceStr = payload.substring(balanceStart, balanceEnd);
            currentUserBalance = balanceStr.toFloat(); // <<< Convert to float

            http.end();
            
            // 💥 NEW: BALANCE CHECK 💥
            float enteredAmount = amount.toFloat();
            if (enteredAmount > currentUserBalance) {
                // Display balance error and reset
                displayError("Insufficient Balance! (Bal: $" + String(currentUserBalance, 2) + ")");
                return false;
            }

            return true; // Success
        } else {
            http.end();
            displayError("User Not Found!");
            return false;
        }
    } else {
        http.end();
        displayError("DB Error: " + String(httpResponseCode));
        return false;
    }
}

/**
 * @brief Sends an HTTP PATCH request to update the user's balance in the 'users' table.
 */
void updateUserBalance(float newBalance) {
    if (WiFi.status() != WL_CONNECTED) {
        Serial.println("Error: Wi-Fi disconnected during balance update.");
        return;
    }
    
    HTTPClient http;
    WiFiClientSecure client;
    client.setInsecure(); // Skip SSL cert verification

    // Use a filter to update only the specific user
    String supabaseUrl = "https://" + String(SUPABASE_HOST) + String(SUPABASE_URL_PREFIX) + "users?id=eq." + currentUserID;
    http.begin(client, supabaseUrl);
    
    // Headers for PATCH
    http.addHeader("Content-Type", "application/json");
    http.addHeader("apikey", SUPABASE_API_KEY);
    http.addHeader("Authorization", "Bearer " + String(SUPABASE_API_KEY));
    http.addHeader("Prefer", "return=minimal"); // Request a minimal response

    // Build JSON body with the new balance (formatted to 2 decimal places)
    String jsonData = "{\"balance\": " + String(newBalance, 2) + "}";

    // Use sendRequest for the PATCH method
    int httpResponseCode = http.sendRequest("PATCH", jsonData); 

    if (httpResponseCode > 0) {
        Serial.println("Balance Update Response Code: " + String(httpResponseCode));
        Serial.println("Balance updated successfully.");
    } else {
        Serial.println("Error updating balance: " + String(httpResponseCode));
    }

    http.end();
}


// ===== NFC Reader (Modified) =====
void readNFC() {
    uint8_t uid[7];
    uint8_t uidLength;

    if (nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, uid, &uidLength)) {
        cardUID = "";
        for (uint8_t i = 0; i < uidLength; i++) {
            // Format UID as uppercase hex string without separators
            if (uid[i] < 0x10) cardUID += "0";
            cardUID += String(uid[i], HEX);
        }
        cardUID.toUpperCase();
        Serial.println("Card UID: " + cardUID);

        tft.fillScreen(TFT_BLACK);
        tft.setCursor(20, 20);
        tft.setTextColor(TFT_YELLOW);
        tft.setTextSize(2);
        tft.println("Card detected. Checking DB...");

        // *** Check DB for card, user data, and perform balance check ***
        if (getCardAndUserData()) { // If this is true, the balance is sufficient
            tft.fillScreen(TFT_BLACK);
            tft.setCursor(20, 20);
            tft.setTextColor(TFT_YELLOW);
            tft.setTextSize(2);
            tft.println("Welcome: " + currentUserName);
            tft.println("Amount: $" + amount);
            tft.print("Balance: $"); 
            tft.println(String(currentUserBalance, 2)); 
            tft.println("\nEnter PIN...");
            delay(1500);
            
            nfcDone = true;
            currentPin = "";
            showPinPad();
        } else {
            // getCardAndUserData() already handles error display and reset
        }
    }
}

void showSuccess(String category) {
    tft.fillScreen(TFT_BLACK);
    tft.setCursor(20, 100);
    tft.setTextColor(TFT_GREEN);
    tft.setTextSize(2);
    tft.print("User: " + currentUserName); // Display fetched user name
    tft.setCursor(20, 130);
    tft.print("Amount: $" + amount);
    tft.setCursor(20, 160);
    tft.print(category);
    tft.println(" Payment Done!");

    // ===== Send Transaction to Supabase (Expense Log) =====
    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        WiFiClientSecure client;
        client.setInsecure(); // Skip SSL cert verification

        String supabaseUrl = "https://" + String(SUPABASE_HOST) + String(SUPABASE_URL_PREFIX) + "transactions";
        http.begin(client, supabaseUrl);

        // Supabase headers
        http.addHeader("Content-Type", "application/json");
        http.addHeader("apikey", SUPABASE_API_KEY);
        http.addHeader("Authorization", "Bearer " + String(SUPABASE_API_KEY));

        // Build JSON body
        String jsonData = "{";
        // *** IMPORTANT: Use the fetched currentUserID ***
        jsonData += "\"user_id\": \"" + currentUserID + "\","; 
        jsonData += "\"type\": \"expense\",";
        jsonData += "\"amount\": " + amount + ",";
        jsonData += "\"category\": \"" + category + "\",";
        jsonData += "\"description\": \"NFC Payment\",";
        jsonData += "\"payment_mode\": \"card\","; // <<< ADDED COMMA
        jsonData += "\"card_id\": \"" + currentCardID + "\""; // <<< NEW: Add the card_id
        jsonData += "}";

        int httpResponseCode = http.POST(jsonData);

        if (httpResponseCode > 0) {
            String response = http.getString();
            Serial.println("Supabase Transaction Response: " + response);

            // 💥 NEW: Calculate and Update Balance 💥
            float enteredAmount = amount.toFloat();
            float newBalance = currentUserBalance - enteredAmount;
            
            Serial.println("Transaction successful. New Balance: " + String(newBalance, 2));
            updateUserBalance(newBalance); // <<< CALL NEW FUNCTION TO PATCH USER TABLE

            // Update success screen with new balance
            tft.setCursor(20, 190);
            tft.setTextColor(TFT_CYAN);
            tft.print("New Balance: $" + String(newBalance, 2));
            
        } else {
            Serial.println("Error sending transaction data: " + String(httpResponseCode));
        }

        http.end();
    }

    // Start the 10-second timer
    successDisplayStartTime = millis();
    displayDateTime();
}
